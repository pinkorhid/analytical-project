WITH ranked_versions AS (
    SELECT
        user_pseudo_id,
        app_info_version,
        ROW_NUMBER() OVER (
            PARTITION BY user_pseudo_id
            ORDER BY event_datetime DESC
        ) AS rn
    FROM events
    WHERE app_info_version IS NOT NULL
),
user_apps AS (
    SELECT
        user_pseudo_id,
        app_info_version
    FROM ranked_versions
    WHERE rn = 1
),
classified_users AS (
    SELECT
        user_pseudo_id,
        CASE
            WHEN app_info_version = '3.1.6'
                THEN 'остання версія (3.1.6)'
            ELSE 'застаріла версія'
        END AS version_group
    FROM user_apps
)
SELECT
    version_group,
    COUNT(*) AS users_count,
    ROUND(
        COUNT(*) * 100.0 /
        (SELECT COUNT(*) FROM classified_users),
        2
    ) AS share_percent
FROM classified_users
GROUP BY version_group;
